   725→function StickyBar({
   726→  house,
   727→  saved,
   728→  onToggleSave,
   729→  onContact,
   730→}: {

   732→  saved: boolean;

   740→    window.addEventListener("scroll", onScroll, { passive: true });
   741→    return () => window.removeEventListener("scroll", onScroll);
   742→  }, []);
   743→  if (!house) return null;
   744→  return (
   745→    <AnimatePresence>

   747→        <motion.div

   758→              </p>

   761→            <div className="hidden sm:flex items-center gap-3 text-sm">
